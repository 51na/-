# 2. 전국 자유 장소 검색 엔진 (Photon API 연동)
# ---------------------------------------------------------
def search_place_candidates(query):
    """
    대한민국 전역을 대상으로 건물, 주소, 상호명 등 모든 텍스트 검색을 지원합니다.
    """
    clean_query = query.strip()
    if not clean_query:
        return []
    
    # 한국 중심점(lat=36.5, lon=127.5) 기준 검색 바이어스 적용 (최대 5개 후보 반환)
    url = f"https://photon.komoot.io/api/?q={clean_query}&lat=36.5&lon=127.5&zoom=7&limit=5"
    candidates = []
    try:
        res = requests.get(url, timeout=5)
        if res.status_code == 200:
            data = res.json()
            for feat in data.get("features", []):
                lon, lat = feat["geometry"]["coordinates"]
                props = feat["properties"]
                name = props.get("name", "")
                street = props.get("street", "")
                city = props.get("city", props.get("state", props.get("county", "")))
                country = props.get("country", "")
                
                # 대한민국 내 결과만 필터링
                if "Korea" in country or "대한민국" in country or "South Korea" in country or not country:
                    parts = [p for p in [name, street, city] if p]
                    display_str = ", ".join(parts) if parts else clean_query
                    candidates.append({
                        "name": display_str,
                        "coord": (lat, lon)
                    })
    except Exception:
        pass
    return candidates

# ---------------------------------------------------------
# 3. OSRM 보행 경로 API
# ---------------------------------------------------------
def get_walking_route(orig_coord, dest_coord):
    url = f"http://router.project-osrm.org/route/v1/foot/{orig_coord[1]},{orig_coord[0]};{dest_coord[1]},{dest_coord[0]}"
    params = {"overview": "full", "geometries": "geojson"}
    try:
        res = requests.get(url, params=params, timeout=5)
        if res.status_code == 200:
            data = res.json()
            if data.get("routes"):
                route = data["routes"][0]
                coords = [[c[1], c[0]] for c in route["geometry"]["coordinates"]]
                return coords, route["distance"]
    except Exception:
        pass
    return None, None
