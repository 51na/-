@st.cache_data(ttl=3600, show_spinner=False)
def get_actual_walk_route(start_coord, end_coord):
    start_coord = tuple(start_coord)
    end_coord = tuple(end_coord)
    try:
        url = (
            f"http://router.project-osrm.org/route/v1/foot/"
            f"{start_coord[1]},{start_coord[0]};{end_coord[1]},{end_coord[0]}"
            f"?overview=full&geometries=geojson"
        )
        res = requests.get(url, timeout=3).json()
        if res.get("code") == "Ok" and res.get("routes"):
            route = res["routes"][0]
            dist_m = int(route["distance"])
            coords = [[pt[1], pt[0]] for pt in route["geometry"]["coordinates"]]
            return coords, dist_m
    except Exception:
        pass

    fallback_dist = int(geodesic(start_coord, end_coord).meters * 1.2)
    return [list(start_coord), list(end_coord)], fallback_dist


def get_multi_waypoint_walk_route(waypoints):
    if len(waypoints) < 2:
        return waypoints, 0

    full_coords = []
    total_dist_m = 0

    for i in range(len(waypoints) - 1):
        seg_coords, seg_dist = get_actual_walk_route(waypoints[i], waypoints[i + 1])
        if i > 0 and full_coords:
            full_coords.extend(seg_coords[1:])
        else:
            full_coords.extend(seg_coords)
        total_dist_m += seg_dist

    return full_coords, total_dist_m


def calc_polyline_distance(coords):
    if len(coords) < 2:
        return 0
    total_m = 0
    for i in range(len(coords) - 1):
        total_m += geodesic(coords[i], coords[i + 1]).meters
    return int(round(total_m))


def build_full_shortcut_waypoints(seg_start, seg_end, middle_points):
    if not middle_points:
        return []

    pts = [list(p) for p in middle_points]
    if geodesic(seg_start, pts[0]).meters > 25:
        pts.insert(0, list(seg_start))
    if geodesic(seg_end, pts[-1]).meters > 25:
        pts.append(list(seg_end))

    return pts


def format_duration(total_seconds, include_sec=True):
    total_seconds = int(round(total_seconds))
    hours = total_seconds // 3600
    minutes = (total_seconds % 3600) // 60
    seconds = total_seconds % 60

    parts = []
    if hours > 0:
        parts.append(f"{hours}시간")
    if minutes > 0 or hours > 0:
        parts.append(f"{minutes}분")
    if include_sec and (seconds > 0 or not parts):
        parts.append(f"{seconds}초")
    return " ".join(parts)

@st.cache_data(ttl=3600, show_spinner=False)
def reverse_geocode_name(lat, lon):
    try:
        url = "https://nominatim.openstreetmap.org/reverse"
        params = {"format": "json", "lat": lat, "lon": lon, "zoom": 18}
        headers = {"User-Agent": "BusanWalkNavigator/1.0"}
        res = requests.get(url, params=params, headers=headers, timeout=2).json()
        name = res.get("name") or res.get("display_name", "").split(",")[0]
        if name:
            return name
    except Exception:
        pass
    return f"위도:{lat:.4f}, 경도:{lon:.4f}"

def find_nearest_station(coord):
    nearest_st = None
    min_dist = float("inf")
    for st_name, st_coord in ALL_STATIONS.items():
        d = geodesic(coord, st_coord).meters
        if d < min_dist:
            min_dist = d
            nearest_st = st_name
    return nearest_st

        )
